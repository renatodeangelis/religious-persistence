\subsection{Mean time to exit}

\begin{figure}[H]
  \centering
  \includegraphics[width=0.85\textwidth]{memory/mte_10yr.png}
  \caption{Mean time to exit (MTE) by origin state across 10-year birth
           cohorts. MTE is the expected number of transmission steps until
           a lineage's religion first differs from its starting state.}
  \label{fig:mte}
\end{figure}

\subsection{$\lambda_2$ trend}

\begin{figure}[H]
  \centering
  \includegraphics[width=0.9\textwidth]{memory/lambda2_trend_10yr.png}
  \caption{$\lambda_2$ (modulus of second-largest eigenvalue of $P_t$) by
           10-year birth cohort, with 95\% bootstrap confidence intervals
           (250 replicates, seed 42). A declining $\lambda_2$ indicates that
           intergenerational religious persistence weakens across cohorts.}
  \label{fig:lambda2_trend}
\end{figure}
